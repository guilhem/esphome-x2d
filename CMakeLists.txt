cmake_minimum_required(VERSION 3.16)
project(esphome_x2d_checks LANGUAGES CXX)
set(X2D_CORE_DIR "" CACHE PATH "Checkout of the component's pinned x2d-core commit")
if(NOT EXISTS "${X2D_CORE_DIR}/src/radio_runtime.h")
  message(FATAL_ERROR "Set X2D_CORE_DIR to the pinned x2d-core checkout")
endif()
enable_testing()
foreach(check IN ITEMS controller radio_encoder)
  add_executable(check_${check} tests/check_${check}.cpp)
  target_compile_features(check_${check} PRIVATE cxx_std_17)
  target_include_directories(check_${check} PRIVATE components/x2d "${X2D_CORE_DIR}/src")
  target_compile_options(check_${check} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME ${check} COMMAND check_${check})
endforeach()
